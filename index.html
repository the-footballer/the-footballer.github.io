-- The Footballer: new card exchange. Trade in as many doubles as you like (at least 1, you keep one of each). -- Every rarity is worth points: Common 10, Uncommon 20, Rare 30, Super Rare 40, Epic 50, Legendary 60, Ultimate 70. -- You get a random card worth the average of your doubles + 10 points (rounded), so about one rarity higher. -- Run after supabase-thursday.sql and supabase-derby-cards.sql. Paste into Supabase > SQL Editor > New snippet, then click Run. -- Safe to run more than once. create or replace function public.exchange_cards(p_players smallint[]) returns jsonb language plpgsql security definer set search_path = '' as $$ declare v_avg numeric; v_tier int; v_target text; v_pid smallint; v_new boolean; g smallint; v_order text[] := array['common','uncommon','rare','super_rare','epic','legendary','ultimate']; begin if auth.uid() is null then raise exception 'Sign in first'; end if; if not public.thursday_open() then raise exception 'The card exchange opens on Thursday at 17:00'; end if; if coalesce(cardinality(p_players), 0) < 1 then raise exception 'Pick at least 1 double'; end if; if (select count(distinct x) from unnest(p_players) x) <> cardinality(p_players) then raise exception 'Pick each double only once'; end if; perform 1 from public.collection c where c.user_id = auth.uid() for update; foreach g in array p_players loop if not exists (select 1 from public.collection c where c.user_id = auth.uid() and c.player_id = g and c.copies >= 2) then raise exception 'You can only exchange doubles (you keep one of each)'; end if; end loop; -- average rarity points of the doubles (Common = 10 ... Ultimate = 70), plus 10 select avg(array_position(v_order, p.rarity) * 10) into v_avg from public.gallery_players p where p.id = any(p_players); v_tier := least(greatest(round((v_avg + 10) / 10)::int, 1), 7); v_target := v_order[v_tier]; update public.collection c set copies = c.copies - 1 where c.user_id = auth.uid() and c.player_id = any(p_players); select p.id into v_pid from public.gallery_players p where p.rarity = v_target and p.special is null and (p.release_at is null or p.release_at <= now()) order by random() limit 1; if v_pid is null then raise exception 'No card of that rarity is available right now'; end if; insert into public.collection as c (user_id, player_id) values (auth.uid(), v_pid) on conflict (user_id, player_id) do update set copies = c.copies + 1 returning (xmax = 0) into v_new; return jsonb_build_object('player_id', v_pid, 'rarity', v_target, 'is_new', v_new, 'avg', round(v_avg, 1)); end $$; revoke execute on function public.exchange_cards(smallint[]) from public, anon; grant execute on function public.exchange_cards(smallint[]) to authenticated;
